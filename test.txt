module pilse_repiter
(
    input logic clk,
    input logic rst_n,
    input logic short_pulse,
    output logic extended_out
);

logic current_state;
logic next_state;
logic [1:0] clk_counter;
logic [1:0] clk_counter_next;

localparam logic STAT_INCREMENT = 1'b0;
localparam logic END_INCREMENT = 1'b1;

always_ff @(posedge clk or negedge rst_n)
begin
if (~rst_n)
begin
    current_state <= STAT_INCREMENT;
    clk_counter <= 2'b00;
end else begin
    current_state <= next_state;
    clk_counter <= clk_counter_next;
end
end

always_comb
begin
    extended_out = 1'b0;
    clk_counter_next = clk_counter;
    next_state = current_state;

    case (current_state)
    STAT_INCREMENT : begin
        clk_counter_next = 2'b00;
        if(short_pulse)
        begin next_state = END_INCREMENT; end
        else begin next_state = STAT_INCREMENT; end
    end

    END_INCREMENT : begin
        clk_counter_next = clk_counter + 1'b1;
        extended_out = 1'b1;
        if(clk_counter != 2'b11)
        begin  next_state = END_INCREMENT; end
        else begin next_state = STAT_INCREMENT; end
    end
    endcase
end



endmodule
