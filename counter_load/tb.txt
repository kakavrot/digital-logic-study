  logic [3:0] tb_out;

  (
    .clk(tb_clk),
    .rst_n(tb_rst_n),
    .load(tb_load),
    .external_data(tb_external_data),
    .out(tb_out)    
  );

    $dumpfile("dump.vcd");

    #10 tb_rst_n = 1;

  #80 $finish;
 end
